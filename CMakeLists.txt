cmake_minimum_required(VERSION 3.25)
project(RifeLove VERSION 0.1.0 LANGUAGES C CXX)

option(RIFELOVE_BUILD_TESTS "Build automated tests" ON)
option(RIFELOVE_BUILD_DESKTOP "Build the Qt Quick desktop shell when Qt is available" ON)
option(RIFELOVE_ALLOW_DEPENDENCY_DOWNLOADS "Allow configure to download build dependencies" OFF)
option(RIFELOVE_BUILD_MEDIA "Build the FFmpeg media engine" OFF)
set(RIFELOVE_FFMPEG_ROOT "$ENV{RIFELOVE_FFMPEG_ROOT}" CACHE PATH "Local FFmpeg SDK root")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(FetchContent)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON)
if(RIFELOVE_ALLOW_DEPENDENCY_DOWNLOADS)
  set(FETCHCONTENT_FULLY_DISCONNECTED OFF)
else()
  set(FETCHCONTENT_FULLY_DISCONNECTED ON)
  if(NOT FETCHCONTENT_SOURCE_DIR_SQLITE3 AND NOT EXISTS "${FETCHCONTENT_BASE_DIR}/sqlite3-src/sqlite3.c")
    message(FATAL_ERROR "Offline configure needs a local SQLite source cache. Seed the cache or explicitly set RIFELOVE_ALLOW_DEPENDENCY_DOWNLOADS=ON for initial setup.")
  endif()
  if(RIFELOVE_BUILD_TESTS AND NOT FETCHCONTENT_SOURCE_DIR_GOOGLETEST AND NOT EXISTS "${FETCHCONTENT_BASE_DIR}/googletest-src/CMakeLists.txt")
    message(FATAL_ERROR "Offline configure needs a local GoogleTest source cache. Seed the cache or explicitly set RIFELOVE_ALLOW_DEPENDENCY_DOWNLOADS=ON for initial setup.")
  endif()
endif()
FetchContent_Declare(sqlite3
  URL https://www.sqlite.org/2025/sqlite-amalgamation-3500400.zip
  URL_HASH SHA256=1d3049dd0f830a025a53105fc79fd2ab9431aea99e137809d064d8ee8356b032
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(sqlite3)
add_library(rifelove_sqlite STATIC ${sqlite3_SOURCE_DIR}/sqlite3.c)
target_include_directories(rifelove_sqlite PUBLIC ${sqlite3_SOURCE_DIR})
target_compile_definitions(rifelove_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DEFAULT_FOREIGN_KEYS=1)

if(RIFELOVE_BUILD_TESTS)
  enable_testing()
  FetchContent_Declare(googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG 52eb8108c5bdec04579160ae17225d66034bd723
    GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(googletest)
endif()

add_library(rifelove_core
  src/common/rational.cpp
  src/timeline/timeline.cpp
  src/database/task_store.cpp)
target_include_directories(rifelove_core PUBLIC src)
target_link_libraries(rifelove_core PUBLIC rifelove_sqlite)
if(MSVC)
  target_compile_options(rifelove_core PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(rifelove_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

add_executable(rifelove-cli apps/cli/main.cpp)
target_link_libraries(rifelove-cli PRIVATE rifelove_core)
if(WIN32)
  target_link_libraries(rifelove-cli PRIVATE shell32)
endif()

if(RIFELOVE_BUILD_MEDIA)
  list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
  find_package(FFmpeg REQUIRED)
  add_library(rifelove_media src/media/ffmpeg_helpers.cpp src/media/probe.cpp src/media/decode.cpp src/media/remux.cpp)
  target_include_directories(rifelove_media PUBLIC src)
  target_link_libraries(rifelove_media PUBLIC rifelove_core PRIVATE FFmpeg::avformat FFmpeg::avcodec FFmpeg::avutil)
  if(MSVC)
    target_compile_options(rifelove_media PRIVATE /W4 /permissive- /EHsc)
  else()
    target_compile_options(rifelove_media PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  target_sources(rifelove-cli PRIVATE apps/cli/media_commands.cpp)
  target_compile_definitions(rifelove-cli PRIVATE RIFELOVE_HAS_MEDIA=1)
  target_link_libraries(rifelove-cli PRIVATE rifelove_media)
  if(WIN32)
    file(GLOB rifelove_ffmpeg_dlls "${RIFELOVE_FFMPEG_ROOT}/bin/*.dll")
    if(rifelove_ffmpeg_dlls)
      file(COPY ${rifelove_ffmpeg_dlls} DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
    endif()
  endif()
endif()

if(RIFELOVE_BUILD_DESKTOP)
  find_package(Qt6 QUIET COMPONENTS Quick QuickControls2)
  if(Qt6_FOUND)
    qt_standard_project_setup()
    qt_add_executable(rifelove-desktop apps/desktop/main.cpp)
    qt_add_qml_module(rifelove-desktop URI RifeLove VERSION 1.0 QML_FILES ui/Main.qml)
    target_link_libraries(rifelove-desktop PRIVATE Qt6::Quick Qt6::QuickControls2 rifelove_core)
  else()
    message(STATUS "Qt 6 Quick was not found; desktop shell is deferred, core and CLI remain available.")
  endif()
endif()

if(RIFELOVE_BUILD_TESTS)
  add_executable(rifelove-tests tests/unit/rational_test.cpp tests/unit/timeline_test.cpp tests/unit/task_store_test.cpp)
  target_link_libraries(rifelove-tests PRIVATE rifelove_core GTest::gtest_main)
  if(RIFELOVE_BUILD_MEDIA)
    target_sources(rifelove-tests PRIVATE tests/unit/media_test.cpp)
    target_link_libraries(rifelove-tests PRIVATE rifelove_media)
  endif()
  include(GoogleTest)
  gtest_discover_tests(rifelove-tests)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME CLI.LocalWorkflow
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/cli_test.py $<TARGET_FILE:rifelove-cli>)
    set_tests_properties(CLI.LocalWorkflow PROPERTIES TIMEOUT 30)
    if(RIFELOVE_BUILD_MEDIA)
      find_program(RIFELOVE_FFMPEG_TOOL ffmpeg HINTS "${RIFELOVE_FFMPEG_ROOT}/bin" REQUIRED)
      find_program(RIFELOVE_FFPROBE_TOOL ffprobe HINTS "${RIFELOVE_FFMPEG_ROOT}/bin" REQUIRED)
      add_test(NAME Media.Integration
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/media_test.py
          $<TARGET_FILE:rifelove-cli> ${RIFELOVE_FFMPEG_TOOL} ${RIFELOVE_FFPROBE_TOOL})
      set_tests_properties(Media.Integration PROPERTIES TIMEOUT 120)
    endif()
  else()
    message(STATUS "Python was not found; CLI integration checks are unavailable (product runtime does not require Python).")
  endif()
endif()
