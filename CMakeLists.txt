cmake_minimum_required(VERSION 3.25)
project(RifeLove VERSION 0.1.0 LANGUAGES C CXX)
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "RifeLove targets Windows x64; select a 64-bit compiler")
endif()

option(RIFELOVE_BUILD_TESTS "Build automated tests" ON)
option(RIFELOVE_BUILD_DESKTOP "Build the Qt Quick desktop shell when Qt is available" ON)
option(RIFELOVE_REQUIRE_DESKTOP "Fail configure if the requested Qt desktop cannot be built" OFF)
option(RIFELOVE_ALLOW_DEPENDENCY_DOWNLOADS "Allow configure to download build dependencies" OFF)
option(RIFELOVE_BUILD_MEDIA "Build the FFmpeg media engine" OFF)
option(RIFELOVE_BUILD_ONNX_CUDA "Build the native ONNX Runtime CUDA backend" OFF)
set(RIFELOVE_FFMPEG_ROOT "$ENV{RIFELOVE_FFMPEG_ROOT}" CACHE PATH "Local FFmpeg SDK root")
set(RIFELOVE_ONNX_ROOT "$ENV{RIFELOVE_ONNX_ROOT}" CACHE PATH "Pinned ONNX Runtime 1.22.0 GPU SDK root")
set(RIFELOVE_JSON_ROOT "$ENV{RIFELOVE_JSON_ROOT}" CACHE PATH "Pinned nlohmann JSON 3.12.0 root")
set(RIFELOVE_TEST_RIFE_PACK "" CACHE PATH "Locally exported and qualified RIFE reference pack for CUDA validation")
set(RIFELOVE_MODEL_TOOLS_PYTHON "" CACHE FILEPATH "Optional development-only PyTorch reference interpreter")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(FetchContent)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON)
if(RIFELOVE_ALLOW_DEPENDENCY_DOWNLOADS)
  set(FETCHCONTENT_FULLY_DISCONNECTED OFF)
else()
  set(FETCHCONTENT_FULLY_DISCONNECTED ON)
  if(NOT FETCHCONTENT_SOURCE_DIR_SQLITE3 AND NOT EXISTS "${FETCHCONTENT_BASE_DIR}/sqlite3-src/sqlite3.c")
    message(FATAL_ERROR "Offline configure needs a local SQLite source cache. Seed the cache or explicitly set RIFELOVE_ALLOW_DEPENDENCY_DOWNLOADS=ON for initial setup.")
  endif()
  if(RIFELOVE_BUILD_TESTS AND NOT FETCHCONTENT_SOURCE_DIR_GOOGLETEST AND NOT EXISTS "${FETCHCONTENT_BASE_DIR}/googletest-src/CMakeLists.txt")
    message(FATAL_ERROR "Offline configure needs a local GoogleTest source cache. Seed the cache or explicitly set RIFELOVE_ALLOW_DEPENDENCY_DOWNLOADS=ON for initial setup.")
  endif()
endif()
FetchContent_Declare(sqlite3
  URL https://www.sqlite.org/2025/sqlite-amalgamation-3500400.zip
  URL_HASH SHA256=1d3049dd0f830a025a53105fc79fd2ab9431aea99e137809d064d8ee8356b032
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(sqlite3)
add_library(rifelove_sqlite STATIC ${sqlite3_SOURCE_DIR}/sqlite3.c)
target_include_directories(rifelove_sqlite PUBLIC ${sqlite3_SOURCE_DIR})
target_compile_definitions(rifelove_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DEFAULT_FOREIGN_KEYS=1)

if(RIFELOVE_BUILD_TESTS)
  enable_testing()
  FetchContent_Declare(googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG 52eb8108c5bdec04579160ae17225d66034bd723
    GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(googletest)
endif()

add_library(rifelove_core
  src/common/rational.cpp
  src/timeline/timeline.cpp
  src/interpolation/scene_analysis.cpp
  src/database/task_store.cpp)
target_include_directories(rifelove_core PUBLIC src)
target_link_libraries(rifelove_core PUBLIC rifelove_sqlite)
if(MSVC)
  target_compile_options(rifelove_core PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(rifelove_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

add_executable(rifelove-cli apps/cli/main.cpp)
target_link_libraries(rifelove-cli PRIVATE rifelove_core)
add_library(rifelove_gpu src/gpu/cuda_device.cpp)
target_include_directories(rifelove_gpu PUBLIC src)
if(MSVC)
  target_compile_options(rifelove_gpu PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(rifelove_gpu PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
target_link_libraries(rifelove-cli PRIVATE rifelove_gpu)
if(WIN32)
  target_link_libraries(rifelove-cli PRIVATE shell32)
endif()

if(RIFELOVE_BUILD_ONNX_CUDA)
  if(NOT WIN32)
    message(FATAL_ERROR "The native ONNX CUDA loader currently targets Windows x64")
  endif()
  find_path(RIFELOVE_ONNX_INCLUDE onnxruntime_c_api.h HINTS "${RIFELOVE_ONNX_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  file(SHA256 "${RIFELOVE_ONNX_INCLUDE}/onnxruntime_c_api.h" rifelove_onnx_header_hash)
  if(NOT rifelove_onnx_header_hash STREQUAL "ed5c8a336708866d21e0c3f5374229e6753088046fcbf69f81394fcbfd730609")
    message(FATAL_ERROR "ONNX Runtime C API header differs from pinned 1.22.0")
  endif()
  find_path(RIFELOVE_JSON_INCLUDE nlohmann/json.hpp HINTS "${RIFELOVE_JSON_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  file(SHA256 "${RIFELOVE_JSON_INCLUDE}/nlohmann/json.hpp" rifelove_json_hash)
  if(NOT rifelove_json_hash STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
    message(FATAL_ERROR "nlohmann JSON header differs from pinned 3.12.0")
  endif()
  add_library(rifelove_inference src/inference/model_pack.cpp src/inference/onnx_cuda_backend.cpp src/inference/tensorrt_cache.cpp)
  target_include_directories(rifelove_inference PUBLIC src PRIVATE "${RIFELOVE_ONNX_INCLUDE}")
  target_include_directories(rifelove_inference SYSTEM PRIVATE "${RIFELOVE_JSON_INCLUDE}")
  target_link_libraries(rifelove_inference PUBLIC rifelove_gpu PRIVATE bcrypt version)
  if(MSVC)
    target_compile_options(rifelove_inference PRIVATE /W4 /permissive- /EHsc)
  else()
    target_compile_options(rifelove_inference PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  add_executable(rifelove-validate-cuda tools/validation/onnx_cuda_validate.cpp)
  target_include_directories(rifelove-validate-cuda SYSTEM PRIVATE "${RIFELOVE_JSON_INCLUDE}")
  target_link_libraries(rifelove-validate-cuda PRIVATE rifelove_inference)
  if(MINGW)
    target_link_options(rifelove-validate-cuda PRIVATE -municode)
  endif()
endif()

if(RIFELOVE_BUILD_MEDIA)
  list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
  find_package(FFmpeg REQUIRED)
  add_library(rifelove_media src/media/ffmpeg_helpers.cpp src/media/probe.cpp src/media/decode.cpp src/media/remux.cpp src/media/output_helpers.cpp src/media/video_writer.cpp src/media/rife_color.cpp)
  target_include_directories(rifelove_media PUBLIC src)
  target_link_libraries(rifelove_media PUBLIC rifelove_core PRIVATE FFmpeg::avformat FFmpeg::avcodec FFmpeg::avutil FFmpeg::swscale)
  if(MSVC)
    target_compile_options(rifelove_media PRIVATE /W4 /permissive- /EHsc)
  else()
    target_compile_options(rifelove_media PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  target_sources(rifelove-cli PRIVATE apps/cli/media_commands.cpp)
  target_compile_definitions(rifelove-cli PRIVATE RIFELOVE_HAS_MEDIA=1)
  target_link_libraries(rifelove-cli PRIVATE rifelove_media)
  if(WIN32)
    file(GLOB rifelove_ffmpeg_dlls "${RIFELOVE_FFMPEG_ROOT}/bin/*.dll")
    if(rifelove_ffmpeg_dlls)
      file(COPY ${rifelove_ffmpeg_dlls} DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
    endif()
  endif()
endif()

if(RIFELOVE_BUILD_MEDIA AND RIFELOVE_BUILD_ONNX_CUDA)
  add_library(rifelove_pipeline src/core/pipeline/video_pipeline.cpp)
  target_include_directories(rifelove_pipeline PUBLIC src)
  target_link_libraries(rifelove_pipeline PUBLIC rifelove_core rifelove_media rifelove_inference)
  if(MSVC)
    target_compile_options(rifelove_pipeline PRIVATE /W4 /permissive- /EHsc)
  else()
    target_compile_options(rifelove_pipeline PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  target_sources(rifelove-cli PRIVATE apps/cli/interpolation_commands.cpp)
  target_compile_definitions(rifelove-cli PRIVATE RIFELOVE_HAS_PIPELINE=1)
  target_link_libraries(rifelove-cli PRIVATE rifelove_pipeline)
  add_library(rifelove_application src/application/processing_service.cpp src/application/local_lease.cpp src/application/job_serialization.cpp src/application/task_queue.cpp src/application/import_service.cpp)
  target_link_libraries(rifelove_application PUBLIC rifelove_pipeline)
  target_include_directories(rifelove_application SYSTEM PRIVATE "${RIFELOVE_JSON_INCLUDE}")
  target_link_libraries(rifelove_application PRIVATE bcrypt)
  target_sources(rifelove-cli PRIVATE apps/cli/queue_commands.cpp)
  target_include_directories(rifelove-cli SYSTEM PRIVATE "${RIFELOVE_JSON_INCLUDE}")
  target_link_libraries(rifelove-cli PRIVATE rifelove_application)
endif()

if(RIFELOVE_BUILD_DESKTOP)
  find_package(Qt6 QUIET COMPONENTS Quick QuickControls2 Multimedia)
  if(Qt6_FOUND)
    qt_standard_project_setup(REQUIRES 6.8)
    qt_add_executable(rifelove-desktop WIN32 apps/desktop/main.cpp)
    set_target_properties(rifelove-desktop PROPERTIES OUTPUT_NAME "RifeLove")
    qt_add_qml_module(rifelove-desktop URI RifeLove VERSION 1.0 QML_FILES ui/Main.qml ui/QueuePage.qml ui/ModelCenterPage.qml ui/ComparisonPage.qml ui/ProgressEstimate.js ui/ActionButton.qml)
    target_link_libraries(rifelove-desktop PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia rifelove_core)
    if(TARGET rifelove_application)
      add_library(rifelove_studio apps/desktop/studio_controller.cpp apps/desktop/studio_controller.hpp
        apps/desktop/queue_controller.cpp apps/desktop/queue_controller.hpp apps/desktop/task_list_model.cpp apps/desktop/task_list_model.hpp)
      target_include_directories(rifelove_studio PUBLIC apps/desktop)
      target_link_libraries(rifelove_studio PUBLIC Qt6::Quick rifelove_application)
      target_link_libraries(rifelove-desktop PRIVATE rifelove_studio)
      target_compile_definitions(rifelove-desktop PRIVATE RIFELOVE_DESKTOP_PROCESSING=1)
    endif()
  else()
    if(RIFELOVE_REQUIRE_DESKTOP)
      message(FATAL_ERROR "The requested Qt desktop SDK was not found")
    endif()
    message(STATUS "Qt 6 Quick was not found; desktop shell is deferred, core and CLI remain available.")
  endif()
endif()

if(RIFELOVE_BUILD_TESTS)
  add_executable(rifelove-tests tests/unit/rational_test.cpp tests/unit/timeline_test.cpp tests/unit/task_store_test.cpp tests/unit/scene_analysis_test.cpp tests/unit/test_runtime_test.cpp)
  target_link_libraries(rifelove-tests PRIVATE rifelove_core GTest::gtest_main)
  if(TARGET rifelove_application)
    target_sources(rifelove-tests PRIVATE tests/unit/application_test.cpp)
    target_include_directories(rifelove-tests SYSTEM PRIVATE "${RIFELOVE_JSON_INCLUDE}")
    target_link_libraries(rifelove-tests PRIVATE rifelove_application)
  endif()
  if(MINGW)
    target_sources(rifelove-tests PRIVATE tests/support/mingw_exit.cpp)
  endif()
  target_sources(rifelove-tests PRIVATE tests/unit/cuda_device_test.cpp)
  target_link_libraries(rifelove-tests PRIVATE rifelove_gpu)
  if(RIFELOVE_BUILD_ONNX_CUDA)
    target_sources(rifelove-tests PRIVATE tests/unit/model_pack_test.cpp tests/unit/tensorrt_cache_test.cpp)
    target_link_libraries(rifelove-tests PRIVATE rifelove_inference)
    if(RIFELOVE_TEST_RIFE_PACK)
      add_test(NAME OnnxCuda.NativeReference COMMAND rifelove-validate-cuda
        "${RIFELOVE_TEST_RIFE_PACK}" "${RIFELOVE_ONNX_ROOT}/lib" "${CMAKE_CURRENT_BINARY_DIR}/native-cuda-validation.json")
      set_tests_properties(OnnxCuda.NativeReference PROPERTIES TIMEOUT 180)
      if(EXISTS "${RIFELOVE_ONNX_ROOT}/lib/nvinfer_10.dll")
        add_test(NAME OnnxTensorRt.NativeReference COMMAND rifelove-validate-cuda
          "${RIFELOVE_TEST_RIFE_PACK}" "${RIFELOVE_ONNX_ROOT}/lib" "${CMAKE_CURRENT_BINARY_DIR}/native-tensorrt-validation.json" tensorrt)
        set_tests_properties(OnnxTensorRt.NativeReference PROPERTIES TIMEOUT 300)
      endif()
    endif()
  endif()
  if(RIFELOVE_BUILD_MEDIA)
    target_sources(rifelove-tests PRIVATE tests/unit/media_test.cpp tests/unit/rife_color_test.cpp)
    target_link_libraries(rifelove-tests PRIVATE rifelove_media)
  endif()
  include(GoogleTest)
  gtest_discover_tests(rifelove-tests)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(TARGET rifelove-desktop)
    add_test(NAME Desktop.Smoke COMMAND rifelove-desktop --smoke-screenshot "${CMAKE_CURRENT_BINARY_DIR}/desktop-smoke.png" --data-dir "${CMAKE_CURRENT_BINARY_DIR}/desktop-smoke-data")
    set_tests_properties(Desktop.Smoke PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QPA_FONTDIR=$ENV{WINDIR}/Fonts")
  endif()
  if(Python3_Interpreter_FOUND)
    add_test(NAME CLI.LocalWorkflow
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/cli_test.py $<TARGET_FILE:rifelove-cli>)
    set_tests_properties(CLI.LocalWorkflow PROPERTIES TIMEOUT 30)
    add_test(NAME Models.SourceIntegrity
      COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests/model_validation -p test_source_integrity.py -v)
    set_tests_properties(Models.SourceIntegrity PROPERTIES TIMEOUT 30)
    if(RIFELOVE_BUILD_MEDIA)
      find_program(RIFELOVE_FFMPEG_TOOL ffmpeg HINTS "${RIFELOVE_FFMPEG_ROOT}/bin" REQUIRED)
      find_program(RIFELOVE_FFPROBE_TOOL ffprobe HINTS "${RIFELOVE_FFMPEG_ROOT}/bin" REQUIRED)
      add_test(NAME Media.Integration
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/media_test.py
          $<TARGET_FILE:rifelove-cli> ${RIFELOVE_FFMPEG_TOOL} ${RIFELOVE_FFPROBE_TOOL})
      set_tests_properties(Media.Integration PROPERTIES TIMEOUT 120)
      if(RIFELOVE_BUILD_ONNX_CUDA AND RIFELOVE_TEST_RIFE_PACK)
        add_executable(rifelove-queue-check tests/integration/queue_workflow.cpp)
        target_link_libraries(rifelove-queue-check PRIVATE rifelove_application)
        if(MINGW)
          target_link_options(rifelove-queue-check PRIVATE -municode)
        endif()
        add_test(NAME Queue.Integration COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/queue_test.py"
          $<TARGET_FILE:rifelove-cli> $<TARGET_FILE:rifelove-queue-check> ${RIFELOVE_FFMPEG_TOOL} ${RIFELOVE_FFPROBE_TOOL}
          "${RIFELOVE_TEST_RIFE_PACK}" "${RIFELOVE_ONNX_ROOT}/lib" "${CMAKE_CURRENT_BINARY_DIR}")
        set_tests_properties(Queue.Integration PROPERTIES TIMEOUT 240)
        if(TARGET rifelove_studio)
          add_executable(rifelove-desktop-check tests/integration/desktop_workflow.cpp)
          target_link_libraries(rifelove-desktop-check PRIVATE rifelove_studio Qt6::QuickControls2)
          add_test(NAME Desktop.Workflow COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/desktop_test.py"
            $<TARGET_FILE:rifelove-desktop-check> ${RIFELOVE_FFMPEG_TOOL} ${RIFELOVE_FFPROBE_TOOL}
            "${RIFELOVE_TEST_RIFE_PACK}" "${RIFELOVE_ONNX_ROOT}/lib" "${CMAKE_CURRENT_SOURCE_DIR}/ui/Main.qml" "${CMAKE_CURRENT_BINARY_DIR}")
          set_tests_properties(Desktop.Workflow PROPERTIES TIMEOUT 210 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QPA_FONTDIR=$ENV{WINDIR}/Fonts")
        endif()
        add_test(NAME Pipeline.Integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/pipeline_test.py
          $<TARGET_FILE:rifelove-cli> ${RIFELOVE_FFMPEG_TOOL} ${RIFELOVE_FFPROBE_TOOL} "${RIFELOVE_TEST_RIFE_PACK}" "${RIFELOVE_ONNX_ROOT}/lib")
        set_tests_properties(Pipeline.Integration PROPERTIES TIMEOUT 180)
        if(RIFELOVE_MODEL_TOOLS_PYTHON)
          add_test(NAME Pipeline.PytorchReference COMMAND "${RIFELOVE_MODEL_TOOLS_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/validation/video_reference.py"
            $<TARGET_FILE:rifelove-cli> ${RIFELOVE_FFMPEG_TOOL} "${RIFELOVE_TEST_RIFE_PACK}" "${RIFELOVE_ONNX_ROOT}/lib" "${CMAKE_CURRENT_BINARY_DIR}/video-reference.json")
          set_tests_properties(Pipeline.PytorchReference PROPERTIES TIMEOUT 180)
        endif()
      endif()
    endif()
  else()
    message(STATUS "Python was not found; CLI integration checks are unavailable (product runtime does not require Python).")
  endif()
endif()
