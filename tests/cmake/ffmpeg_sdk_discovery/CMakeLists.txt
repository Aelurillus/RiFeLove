cmake_minimum_required(VERSION 3.24)
project(RifeLoveFFmpegDiscoveryRegression LANGUAGES NONE)
list(PREPEND CMAKE_MODULE_PATH "${RIFELOVE_MODULE_ROOT}")
set(CMAKE_FIND_LIBRARY_PREFIXES "lib" "")
set(CMAKE_FIND_LIBRARY_SUFFIXES ".dll.a" ".lib" ".a")
find_package(FFmpeg REQUIRED)
if(RIFELOVE_SECOND_SDK)
  set(RIFELOVE_FFMPEG_ROOT "${RIFELOVE_SECOND_SDK}")
  find_package(FFmpeg REQUIRED)
endif()
file(WRITE "${CMAKE_BINARY_DIR}/resolved-targets.txt" "${FFmpeg_INCLUDE_DIR}\n")
foreach(component avformat avcodec avutil swscale)
  get_target_property(location FFmpeg::${component} IMPORTED_LOCATION)
  get_target_property(includes FFmpeg::${component} INTERFACE_INCLUDE_DIRECTORIES)
  file(APPEND "${CMAKE_BINARY_DIR}/resolved-targets.txt" "${location}\n${includes}\n")
endforeach()
